```
│ Authorization: Bearer {jwt_token}       │
└─────────────────────────────────────────┘
```

### 6.2 RLS (Row Level Security) 設計

```sql
-- 会社単位でのデータ分離
CREATE POLICY "company_isolation" ON customers
  USING (company_id = auth.jwt() -> 'user_metadata' ->> 'company_id');

-- スタッフは自社データのみアクセス可
CREATE POLICY "staff_access" ON sales
  USING (
    EXISTS (
      SELECT 1 FROM staff
      WHERE staff.user_id = auth.uid()
      AND staff.company_id = sales.company_id
    )
  );
```
